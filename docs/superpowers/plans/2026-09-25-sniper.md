> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** เพิ่มปืนสไนเปอร์ (สโคป + ยิงหลังรอ 0.5 วิ + ลด 50% เลือด + กินมานา) ทั้งเกมหลักและ PVP, แถบดีเลย์ให้ทุกท่าที่รอ ≥0.3 วิ, ร้าน PVP แบ่งเล็ก/กลาง/ใหญ่, ยิงหัวใน PVP, แบนเนอร์คิลใน PVP

```lua
-- node tools/lua.js tests/sniper_config_test.lua Edit
-- ข้อมูลของสไน/แถบดีเลย์/ร้าน PVP ตามสเปก docs/superpowers/specs/2026-09-25-sniper-design.md
local function fresh(module)
	local copy = module:Clone()
	copy.Name = module.Name .. "_t" .. math.random(1, 1e9)
	copy.Parent = module.Parent
	local ok, result = pcall(require, copy)
	copy:Destroy()
	assert(ok, result)
	return result
end

local RS = game.ReplicatedStorage
local ItemCatalog = fresh(RS.ItemCatalog)
local CombatConfig = fresh(RS.CombatConfig)
local CraftConfig = fresh(RS.CraftConfig)
local ArenaConfig = fresh(RS.ArenaConfig)
local out, fails = {}, 0
local function check(ok, text)
	out[#out + 1] = (ok and "ok   " or "FAIL ") .. text
	if not ok then fails += 1 end
end

local item = ItemCatalog.Items.Sniper
check(item and item.kind == "weapon" and item.icon ~= "", "ไอเทม Sniper เป็นอาวุธ มีไอคอน")
local w = CombatConfig.Weapons.Sniper
check(w and w.quickDraw and w.scope and w.manaCost == 1, "สไน: คลิกเดียว สโคป มานา 1")
check(w and w.pierceFraction == 0.5 and w.mobDamage == 150, "สไน: คน 50% · มอน 150")
local a = w and w.attacks[1]
check(a and a.windup == 0.5 and a.projectile and a.projectile.kind == "hitscan", "สไน: รอ 0.5 วิ ยิงแบบ hitscan")
check(CombatConfig.Sounds.Sniper and CombatConfig.Sounds.Sniper.cast == "rbxassetid://101482767943742", "เสียงยิงสไน")

local recipe
for _, r in ipairs(CraftConfig.Recipes or CraftConfig) do
	if type(r) == "table" and r.id == "Sniper" and r.tab == "weapon" then recipe = r end
end
check(recipe ~= nil, "สูตรคราฟสไนที่ช่างตีเหล็ก")
for id in pairs(recipe and recipe.cost or {}) do
	check(ItemCatalog.Items[id] ~= nil, "วัสดุ " .. id .. " มีจริง")
end

-- แถบดีเลย์: ท่าที่รอ >= 0.3 วิ เท่านั้น
check(CombatConfig.ChargeBarMinSeconds == 0.3, "เกณฑ์แถบ 0.3 วิ")
local function charged(id)
	return CombatConfig.Weapons[id].attacks[1].windup >= CombatConfig.ChargeBarMinSeconds
end
for _, id in ipairs({ "Greatsword", "Pistol", "Bow", "Sniper" }) do check(charged(id), id .. " มีแถบ") end
for _, id in ipairs({ "Sword", "Staff", "Spellbook", "Fists" }) do check(not charged(id), id .. " ไม่มีแถบ") end

-- ร้าน PVP
local sn, gs = ArenaConfig.ShopEntry("Sniper"), ArenaConfig.ShopEntry("Greatsword")
check(sn and sn.price == 3000 and sn.size == "large" and sn.row == "ranged" and sn.fillMana == true, "ร้าน: สไน 3,000 ใหญ่ มานาเต็ม")
check(gs and gs.price == 3000 and gs.size == "large", "ร้าน: ดาบใหญ่ 3,000 ใหญ่")
local sizes = { Sword = "small", Shield = "medium", Pistol = "small", Bow = "medium", Staff = "medium", Spellbook = "medium" }
for id, size in pairs(sizes) do
	check(ArenaConfig.ShopEntry(id).size == size, ("ร้าน: %s = %s"):format(id, size))
end
for _, e in ipairs(ArenaConfig.Shop) do
	if e.slot == "weapon" then check(ArenaConfig.SizeOrder[e.size] ~= nil, e.id .. " มีขนาด") end
end
check(ArenaConfig.HeadshotMultiplier and ArenaConfig.HeadshotMultiplier > 1, "ยิงหัวแรงกว่า")
check(ArenaConfig.MultiKillSeconds == 10 and ArenaConfig.KillSound == "rbxassetid://107279082816485", "คิลต่อเนื่อง 10 วิ + เสียงคิล")

local remotes = RS.Remotes.Source
check(remotes:find("\"RequestAimUpdate\"") and remotes:find("\"ArenaKillBanner\""), "remote ใหม่สองตัว")

return (fails == 0 and "PASS\n" or ("FAIL x%d\n"):format(fails)) .. table.concat(out, "\n")
```

- [ ] **Step 5: PlayerCombat — fireSniper** วางต่อจาก `fireProjectile`:

```lua
-- สไน: raycast ครั้งเดียวถึงเป้าทันที (ไม่ใช่วงกว้าง 8 สตัดแบบกระสุนอื่น — สไนต้องเล็งจริง)
-- คน: ลด `pierceFraction` ของเลือดเต็ม ข้ามเกราะ · หัว (สนามเท่านั้น) = เลือดที่เหลือทั้งหมด
-- มอนเกมหลัก: `mobDamage` คงที่ · บอทในสนาม = เป็นฝั่งตรงข้าม ไม่ใช่มอน จึงคิดแบบเดียวกับคน
local function fireSniper(player, weapon, attack, aim)
	local spec = attack.projectile
	local character = player.Character
	local hrp = character and character:FindFirstChild("HumanoidRootPart")
	if not hrp then return end
	local muzzle = hrp.Position + Vector3.new(0, spec.muzzleHeight or 1.5, 0)
	local forward = aimDirection(hrp, muzzle, aim)

	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	params.FilterDescendantsInstances = { character }
	params.IgnoreWater = true
	local result = workspace:Raycast(muzzle, forward * spec.length, params)
	local endPoint = result and result.Position or (muzzle + forward * spec.length)

	playCastSound(player) -- เสียงปืนดังตอนกระสุนออก ไม่ใช่ตอนคลิก

	-- เส้นกระสุนวาบให้ทุกคนเห็นว่ามีคนยิงมาจากไหน
	local tracer = Instance.new("Part")
	tracer.Name = "WOLD_Projectile"
	tracer.Anchored, tracer.CanCollide, tracer.CanQuery, tracer.CanTouch = true, false, false, false
	local distance = (endPoint - muzzle).Magnitude
	tracer.Size = Vector3.new(spec.size.X, spec.size.Y, math.max(0.1, distance))
	tracer.CFrame = CFrame.lookAt(muzzle + (endPoint - muzzle) / 2, endPoint)
	tracer.Color, tracer.Material = spec.color, spec.material
	tracer.Parent = workspace
	Debris:AddItem(tracer, spec.life)

	if not result then return end
	local model = characterOf(result.Instance)
	if not model then return end
	local head = headOwner(result.Instance) == model
	-- ทีม/ตี้เดียวกัน = ทะลุไปเหมือนไม่มีใคร (ทั้งคนและบอท)
	if ArenaAccess.Friendly and ArenaAccess.Friendly(player, Players:GetPlayerFromCharacter(model) or model) then return end

	local victim = Players:GetPlayerFromCharacter(model)
	if victim then
		if victim == player or Party.SameParty(player, victim) or Vitals.IsInvulnerable(victim) then return end
		local vstate = Vitals.Get(victim)
		if not vstate then return end
		local amount = head and vstate.hp or Vitals.MaxHP(victim) * weapon.pierceFraction
		if head and ArenaAccess.OnHeadshot then ArenaAccess.OnHeadshot(player, victim) end
		local wasDowned = Vitals.IsDowned(victim)
		local landed = Vitals.ApplyDamage(victim, amount, "combat", true)
		if landed and ArenaAccess.OnHit then ArenaAccess.OnHit(player, victim, amount) end
		reportHit(player, victim)
		playHitSound(model, player)
		if not wasDowned and Vitals.IsDowned(victim) then
			awardKill(player, victim)
			StatusEffects.SetAsleep(victim, true)
		end
		return
	end

	if CollectionService:HasTag(model, "WOLD_Dummy") then
		Progression.AwardExp(player, ProgressionConfig.ExpPerDummyHit)
		playHitSound(model, player)
		return
	end
	if CollectionService:HasTag(model, "WOLD_Mob") and MobAccess.Damage then
		local amount = weapon.mobDamage
		local humanoid = model:FindFirstChildOfClass("Humanoid")
		if ServerConfig.IsArena and humanoid then
			amount = head and humanoid.Health or humanoid.MaxHealth * weapon.pierceFraction
			if head and ArenaAccess.OnHeadshot then ArenaAccess.OnHeadshot(player, model) end
		end
		if MobAccess.Damage(model, amount, player, true) then playHitSound(model, player) end
	end
end
```
