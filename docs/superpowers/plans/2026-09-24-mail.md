# ระบบเมล (กล่องจดหมายในหินเวท) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** กล่องจดหมายในหินเวทสื่อสาร ที่ระบบ แอดมิน และเพื่อนส่งของถึงกันได้ ทั้งตอนออฟไลน์และข้ามเซิร์ฟ โดยของไม่หายและไม่ซ้ำ

**Architecture:** เมลเดินทางด้วย `ProfileStore:MessageAsync` (ผ่าน `ProfileAccess.MessageAsync` ที่มีอยู่แล้ว) เข้าไปเก็บใน `profile.Data.Mail` ของผู้รับ `Mail.lua` เป็นโมดูลที่ไม่รู้จัก remote/UI ทำหน้าที่ส่ง รับเข้ากล่อง และถือคิวตอนกล่องเต็ม `PlayerMail.server.lua` เป็นคนตัดสินกฎของผู้เล่น (สหาย ค่าส่ง ของต้องห้าม) และคุย remote · UI เป็นหน้าใหม่ใน `SocialGui`

**Tech Stack:** Luau บน Roblox · ProfileStore (vendored) · Rojo · Studio MCP (`tools/lua.js`, `tools/syntax.js`, `tools/push.js`, `tools/mcp-call.js`)

**Spec:** `docs/superpowers/specs/2026-09-24-mail-design.md`

## Global Constraints

- ค่าส่ง **500 ซิลเวอร์** (`Silver` เป็นไอเทม ไม่ใช่เงิน) ต่อฉบับ · ผู้เล่นแนบของ **1 ช่อง** · **แนบเงินไม่ได้**
- ผู้เล่นส่งได้เฉพาะคนใน **รายชื่อสหาย** (`Friends.Get(player).friends[userId]`)
- ส่งไม่ได้: `bound` · `kind == "key"` · ไอเทมที่มี `use` · ของใน `CraftConfig.Recipes` · `Chemical` `GearSpare` `Blueprint` `Insignia` `Silver`
- กล่องจุ **50 ฉบับ** · **ไม่หมดอายุ** · เต็มแล้วเมลรอคิว ไม่หาย ค่าส่งไม่คืน
- เมลจากระบบ/แอดมินไม่ติดข้อห้ามของผู้เล่น แนบของได้หลายชิ้น + `gold` + `exp`
- UI เป็น **หน้าเมนูในหินเวทสื่อสาร** (`SocialGui`) ไม่ใช่หน้าต่างแยก
- ข้อความจากผู้เล่นต้องผ่าน `TextService` ก่อนเก็บ (กฎของ Roblox สำหรับข้อความที่คนอื่นอ่าน)
- ข้อความ UI/คอมเมนต์เป็นภาษาไทยตามโค้ดรอบข้าง · ใช้ TAB เยื้องย่อหน้า
- ⚠️ **เซิร์ฟสนาม (Arena) ห้ามเปิดเซสชันโปรไฟล์** — `PlayerMail.server.lua` ต้องอยู่ใน `globIgnorePaths` ของ `arena.project.json`
- ⚠️ `require` ใน Edit mode แคชผลข้าม `execute_luau` → เทส Edit ต้อง Clone โมดูลไว้ที่เดิมแล้ว require ตัวโคลน
- ⚠️ `require` ใน sandbox ของ Play ได้สำเนาใหม่ของโมดูล (ตารางว่าง) → เทส Play ต้องผ่าน `ServerStorage.MailTest` (BindableFunction, Studio เท่านั้น)

## Review Focus

1. **กดส่งรัวระหว่างที่ MessageAsync ยังไม่ตอบ** — ต้องหักของ/ค่าส่งครั้งเดียว ครั้งที่สองถูกปฏิเสธ (เทส `DoubleSend` ใน Task 3)
2. **ของในช่องเปลี่ยนระหว่างรอกรองข้อความ** (yield) — `send` ตรวจช่องซ้ำหลัง yield ก่อนหัก (โค้ดใน Task 3 · ครอบด้วย `DoubleSend`)
3. **กดรับตอนกระเป๋าเต็ม** — ปฏิเสธ เมลยังไม่ถูก `claimed` ของไม่หาย (เทสใน Task 3)
4. **ไอเทมในเมลถูกถอดออกจากแคตตาล็อก** — รับส่วนที่ยังมีได้ ไม่ติดทั้งฉบับ (เทสใน Task 3)
5. **กล่องเต็มแล้วมีเมลเข้ามา** — รอคิว ไม่เรียก `processed()` ลบหนึ่งฉบับแล้วดึงเข้ากล่อง (เทสใน Task 2 และ `FullBox` ใน Task 3)

-- ผู้เล่นส่งของชิ้นนี้ทางเมลได้ไหม (ผู้ใช้กำหนด 2026-09-24) — ด่านเดียว ทั้งปุ่มฝั่งไคลเอนต์และเซิร์ฟเวอร์ถามตัวนี้
--
-- ที่เหลือให้ส่งคือวัตถุดิบ: ของที่กดใช้ได้ ของที่คราฟได้ และของต้องห้ามใน MailConfig ทำให้ร้าน/โรงตีเหล็ก/
-- ฟาร์มเถื่อนยังเป็นทางเดียวที่ของพวกนั้นเข้าเกม · เมลจากระบบ/แอดมินไม่ผ่านด่านนี้
function ItemCatalog.IsMailable(id)
	local item = ItemCatalog.Items[id]
	if not item then return false end
	if item.bound or item.kind == "key" then return false end
	if item.use then return false end
	if CraftableOutput[id] then return false end
	return not MailConfig.Contraband[id]
end
```

  - `Mail.Find(data, letterId) -> (index?, letter?)` · `Mail.HasAttachment(letter) -> boolean` · `Mail.Unread(data) -> number`
  - `Mail.Changed : ((player, letter?) -> ())?` — hook ที่ PlayerMail ตั้ง
  - รูปทรงจดหมายหลัง sanitize: `{ id, at, from, fromName, subject, body, items = {{id, count, durability?, meta?}}, gold, exp, claimed, read }`

-- node tools/lua.js tests/mail_core_test.lua Edit
--
-- ตรรกะกล่องจดหมายล้วนๆ บนตาราง data ปลอม ไม่แตะโปรไฟล์จริง ไม่แตะ DataStore (Transport ถูกสวมแทน)

	local copy = module:Clone()
	copy.Name = module.Name .. "_t" .. math.random(1, 1e9)
	copy.Parent = module.Parent
	local ok, result = pcall(require, copy)
	copy:Destroy()
	assert(ok, result)
	return result
end

local Mail = fresh(game.ServerScriptService.Mail)
local out, fails = {}, 0
local function check(ok, text)
	out[#out + 1] = (ok and "ok   " or "FAIL ") .. text
	if not ok then fails += 1 end
end

-- Send: เติม kind/id/at แล้วยื่นให้ Transport
local sentTo, sentLetter
Mail.Transport = function(userId, letter) sentTo, sentLetter = userId, letter return true end
check(Mail.Send(123, { fromName = "เทส", items = {} }) == true, "Send คืน true")
check(sentTo == 123 and sentLetter.kind == "mail", "Transport ได้ userId และ kind=mail")
check(type(sentLetter.id) == "string" and #sentLetter.id > 10 and type(sentLetter.at) == "number", "มี id และ at")
Mail.Transport = function() return false end
check(Mail.Send(123, {}) == false, "Transport ล้ม = Send คืน false")

-- Deliver ลงกล่อง
local player, data = {}, {}
local called = 0
local function spy() called += 1 end
check(Mail.Deliver(player, data, { id = "a", items = { { id = "Stone", count = "3" } }, gold = 5 }, spy) == true, "Deliver เข้ากล่อง")
check(called == 1 and #Mail.Box(data) == 1, "processed ถูกเรียก กล่องมี 1")
local _, a = Mail.Find(data, "a")
check(a and a.items[1].count == 3 and a.gold == 5 and a.claimed == false, "sanitize: count เป็นตัวเลข gold ติดมา")
check(Mail.HasAttachment(a) and Mail.Unread(data) == 1, "มีของแนบ ยังไม่อ่าน 1")

-- id ซ้ำ = ไม่เพิ่ม แต่ปล่อยคิว
Mail.Deliver(player, data, { id = "a" }, spy)
check(called == 2 and #Mail.Box(data) == 1, "id ซ้ำ: processed แต่ไม่ซ้ำในกล่อง")

-- ของเสีย = ทิ้ง (ไม่งั้นค้างคิวตลอดไป)
check(Mail.Deliver(player, data, "ขยะ", spy) == true and called == 3 and #Mail.Box(data) == 1, "ข้อความเสียถูกทิ้ง")

-- กล่องเต็ม = รอคิว ไม่เรียก processed
for i = 2, 50 do Mail.Deliver(player, data, { id = "f" .. i }, function() end) end
check(#Mail.Box(data) == 50, "เติมจนเต็ม 50")
local lateCalled = false
check(Mail.Deliver(player, data, { id = "late" }, function() lateCalled = true end) == false, "เต็มแล้ว Deliver คืน false")
check(lateCalled == false and Mail.HeldCount(player) == 1 and not Mail.Find(data, "late"), "รอคิว ไม่ processed ไม่อยู่ในกล่อง")

-- ลบหนึ่งฉบับแล้ว Drain = ดึงเข้ากล่อง
table.remove(Mail.Box(data), 1)
Mail.Drain(player, data)
check(lateCalled == true and Mail.HeldCount(player) == 0 and Mail.Find(data, "late") ~= nil, "Drain ดึงคิวเข้ากล่อง")
check(#Mail.Box(data) == 50, "ยัง 50 พอดี")

-- Forget ล้างคิว (คนออกเกม — ProfileStore ยังเก็บข้อความที่ไม่ processed ไว้ให้รอบหน้า)
Mail.Deliver(player, data, { id = "late2" }, function() end)
Mail.Forget(player)
check(Mail.HeldCount(player) == 0, "Forget ล้างคิวในหน่วยความจำ")

return (fails == 0 and "PASS\n" or ("FAIL x%d\n"):format(fails)) .. table.concat(out, "\n")

-- ดีไซน์: docs/superpowers/specs/2026-09-24-mail-design.md
--
-- เมลเดินทางด้วย ProfileStore:MessageAsync เข้าไปอยู่ในโปรไฟล์ของผู้รับเอง ส่งถึงคนที่ออฟไลน์หรือเล่นอยู่
-- อีกเซิร์ฟได้ และตอนกดรับ เมลกับกระเป๋าอยู่ในโปรไฟล์ก้อนเดียวกัน = ของไม่หายไม่ซ้ำ
--
-- ⚠️ **ข้อความที่ยังไม่ `processed()` ProfileStore เก็บค้างไว้ในโปรไฟล์** แต่เรียกตัวรับให้แค่ครั้งเดียวต่อเซสชัน
-- กล่องเต็มจึงต้องจำ callback ไว้เอง (Held) แล้วดึงเข้ากล่องเมื่อมีช่อง · เซิร์ฟปิดก่อน = มาใหม่รอบหน้า ไม่หาย
local HttpService = game:GetService("HttpService")
local MailConfig = require(game.ReplicatedStorage.MailConfig)
local ProfileAccess = require(script.Parent.ProfileAccess)

local Mail = {}

local Held = {} -- [player] = { { letter, processed }, ... } เมลที่มาถึงตอนกล่องเต็ม
local fallbackStore = nil

-- hook: PlayerMail.server ตั้งไว้เพื่อส่งภาพกล่องใหม่ให้ไคลเอนต์ · letter = ฉบับที่เพิ่งมาถึง (nil = แค่เปลี่ยน)
Mail.Changed = nil

-- ทางส่งจริง แยกออกมาให้เทสสวมแทนได้โดยไม่แตะ DataStore
function Mail.Transport(userId, letter)
	if ProfileAccess.MessageAsync then return ProfileAccess.MessageAsync(userId, letter) end
	-- ponytail: เซิร์ฟสนามไม่มี PlayerProfile.server (ห้ามเปิดเซสชันที่นั่น) จึงไม่มีใครตั้ง MessageAsync
	-- เปิด store ชื่อเดียวกันเองไว้ส่งข้อความอย่างเดียว — MessageAsync ไม่ต้องเปิดเซสชัน ไม่แย่งล็อกใคร
	fallbackStore = fallbackStore or require(script.Parent.ProfileStore).New("WOLD_PlayerProfile_v1", {})
	local ok, sent = pcall(function()
		return fallbackStore:MessageAsync("Player_" .. userId, letter)
	end)
	if not ok then warn("[Mail] ส่งถึง", userId, "ไม่ได้:", sent) end
	return ok and sent == true
end

-- ส่งเมลหนึ่งฉบับ คืน true เมื่อ ProfileStore รับฝากแล้ว (ยัง yield ได้หลายวินาที)
function Mail.Send(toUserId, letter)
	toUserId = tonumber(toUserId)
	if not toUserId or type(letter) ~= "table" then return false end
	letter.kind = "mail"
	letter.id = HttpService:GenerateGUID(false)
	letter.at = os.time()
	return Mail.Transport(toUserId, letter) == true
end

function Mail.Box(data)
	if type(data.Mail) ~= "table" then data.Mail = {} end
	return data.Mail
end

function Mail.Find(data, letterId)
	for index, letter in ipairs(Mail.Box(data)) do
		if letter.id == letterId then return index, letter end
	end
	return nil, nil
end

function Mail.HasAttachment(letter)
	return #(letter.items or {}) > 0 or (letter.gold or 0) > 0 or (letter.exp or 0) > 0
end

function Mail.Unread(data)
	local n = 0
	for _, letter in ipairs(Mail.Box(data)) do
		if not letter.read then n += 1 end
	end
	return n
end

-- ข้อความมาจากเซิร์ฟของเราเองทั้งหมด แต่ผ่าน DataStore มา — จัดรูปให้ตรงก่อนเก็บ ของที่ผิดรูปคือเมลที่
-- จะทำให้หน้าจอหรือการกดรับพังทุกครั้งที่เปิดกล่อง
local function sanitize(m)
	if type(m) ~= "table" or type(m.id) ~= "string" then return nil end
	local items = {}
	for _, it in ipairs(type(m.items) == "table" and m.items or {}) do
		if type(it) == "table" and type(it.id) == "string" then
			items[#items + 1] = {
				id = it.id,
				count = math.max(1, math.floor(tonumber(it.count) or 1)),
				durability = tonumber(it.durability),
				meta = type(it.meta) == "table" and it.meta or nil,
			}
		end
	end
	return {
		id = m.id,
		at = tonumber(m.at) or os.time(),
		from = tonumber(m.from) or 0,
		fromName = tostring(m.fromName or "ระบบ"),
		subject = tostring(m.subject or ""),
		body = tostring(m.body or ""),
		items = items,
		gold = math.max(0, math.floor(tonumber(m.gold) or 0)),
		exp = math.max(0, math.floor(tonumber(m.exp) or 0)),
		claimed = m.claimed == true,
		read = m.read == true,
	}
end

-- ตัวรับข้อความ `kind == "mail"` (เรียกจาก MessageHandler ใน PlayerProfile.server)
function Mail.Deliver(player, data, message, processed)
	local letter = sanitize(message)
	-- ข้อความเสีย: ทิ้ง ถ้าไม่ processed มันจะค้างคิวแล้วมาหาทุกครั้งที่เข้าเกม
	if not letter then processed() return true end
	-- id ซ้ำ = ฉบับนี้อยู่ในกล่องแล้ว (ProfileStore ส่งซ้ำได้ถ้าเซิร์ฟล่มก่อนเซฟ) ปล่อยคิวเฉยๆ
	if Mail.Find(data, letter.id) then processed() return true end

	local box = Mail.Box(data)
	if #box >= MailConfig.BoxSize then
		Held[player] = Held[player] or {}
		table.insert(Held[player], { letter = letter, processed = processed })
		return false
	end
	table.insert(box, letter)
	processed()
	if Mail.Changed then task.defer(Mail.Changed, player, letter) end
	return true
end

-- ดึงเมลที่รอคิวเข้ากล่องเท่าที่มีช่อง เรียกทุกครั้งที่กล่องมีช่องว่างเพิ่ม (ลบ/ล้าง) และตอนเข้าเกม
function Mail.Drain(player, data)
	local held = Held[player]
	while held and #held > 0 and #Mail.Box(data) < MailConfig.BoxSize do
		local entry = table.remove(held, 1)
		Mail.Deliver(player, data, entry.letter, entry.processed)
	end
end

function Mail.HeldCount(player)
	return Held[player] and #Held[player] or 0
end

function Mail.Forget(player)
	Held[player] = nil
end

return Mail
```

-- ดีไซน์: docs/superpowers/specs/2026-09-24-mail-design.md · ตรรกะกล่องอยู่ที่ Mail.lua ไฟล์นี้ถือแค่กฎของผู้เล่น

if ServerConfig.IsLobby or ServerConfig.IsArena then return end

local TextService = game:GetService("TextService")
local Mail = require(game.ServerScriptService.Mail)
local MailConfig = require(game.ReplicatedStorage.MailConfig)
local ItemCatalog = require(game.ReplicatedStorage.ItemCatalog)
local Inventory = require(game.ServerScriptService.Inventory)
local Friends = require(game.ServerScriptService.Friends)
local Progression = require(game.ServerScriptService.Progression)

-- [player] = true ระหว่างส่ง · MessageAsync ค้างได้หลายวินาที ถ้าไม่กัน กดสองทีหักของสองรอบ
local Sending = {}

local function snapshot(player)
	local data = ProfileAccess.Get(player)
	if not data then return end
	local unread = Mail.Unread(data)
	player:SetAttribute("WOLD_MailUnread", unread)
	Remotes.MailUpdate:FireClient(player, {
		letters = Mail.Box(data), cap = MailConfig.BoxSize, fee = MailConfig.Fee, unread = unread,
	})
end

Mail.Changed = function(player, letter)
	if not player:IsDescendantOf(Players) then return end
	snapshot(player)
	if letter then
		Remotes.SocialNotify:FireClient(player, "mail", ("มีจดหมายใหม่จาก %s"):format(letter.fromName))
	end
end

-- ข้อความที่คนอื่นจะอ่านต้องผ่านตัวกรองของ Roblox · Broadcast เพราะผู้รับอ่านทีหลัง อาจอยู่คนละเซิร์ฟ
local function filtered(player, text)
	if text == "" then return "" end
	local ok, result = pcall(function()
		return TextService:FilterStringAsync(text, player.UserId, Enum.TextFilterContext.PrivateChat)
			:GetNonChatStringForBroadcastAsync()
	end)
	return ok and result or nil
end

local function send(player, toUserId, slotIndex, count, body)
	if Sending[player] then return false, "กำลังส่งฉบับก่อนหน้า รอสักครู่" end
	toUserId, slotIndex = tonumber(toUserId), tonumber(slotIndex)
	count = math.floor(tonumber(count) or 0)
	local mine = Friends.Get(player)
	local friendName = toUserId and mine and mine.friends[toUserId]
	if not friendName then return false, "ส่งได้เฉพาะคนในรายชื่อสหาย" end

	local state = Inventory.Get(player)
	local slot = state and slotIndex and state.slots[slotIndex]
	if not slot then return false, "ไม่มีของในช่องนั้น" end
	if not ItemCatalog.IsMailable(slot.id) then return false, "ของชิ้นนี้ส่งทางเมลไม่ได้ (ส่งได้เฉพาะวัตถุดิบ)" end
	if count < 1 or count > (slot.count or 1) then return false, "จำนวนไม่ถูกต้อง" end
	if Inventory.CountOf(player, MailConfig.FeeItem) < MailConfig.Fee then
		return false, ("ค่าส่ง %d ซิลเวอร์ — ซิลเวอร์ไม่พอ"):format(MailConfig.Fee)
	end

	Sending[player] = true
	local ok, result = pcall(function()
		body = type(body) == "string" and body:gsub("[\r\n\t]", " "):sub(1, MailConfig.BodyMaxLength) or ""
		local clean = filtered(player, body)
		if not clean then return { false, "กรองข้อความไม่สำเร็จ ลองใหม่อีกครั้ง" } end

		-- ตัวกรอง yield → ของในช่องอาจเปลี่ยนไปแล้ว (ใช้ ทิ้ง ย้าย) ตรวจใหม่ก่อนหัก
		slot = state.slots[slotIndex]
		if not slot or not ItemCatalog.IsMailable(slot.id) or count > (slot.count or 1) then
			return { false, "ของในช่องเปลี่ยนไประหว่างส่ง" }
		end
		if not Inventory.RemoveById(player, MailConfig.FeeItem, MailConfig.Fee) then
			return { false, "ซิลเวอร์ไม่พอ" }
		end
		local item = { id = slot.id, count = count, durability = slot.durability, meta = slot.meta }
		Inventory.RemoveFromSlot(player, slotIndex, count)

		local sent = Mail.Send(toUserId, {
			from = player.UserId, fromName = player.DisplayName, subject = "", body = clean, items = { item },
		})
		if not sent then
			-- ponytail: คืนเข้ากระเป๋าตรงๆ ช่องที่เพิ่งว่างถูกเติมเต็มระหว่าง yield ได้ในทางทฤษฎี (ของหาย)
			-- ถ้าเจอจริง ให้ส่งคืนเป็นเมลถึงตัวเองแทน
			Inventory.AddItem(player, item.id, item.count, item.durability, item.meta)
			Inventory.AddItem(player, MailConfig.FeeItem, MailConfig.Fee)
			return { false, "ส่งไม่สำเร็จ (ไปรษณีย์ไม่ตอบ) คืนของและค่าส่งแล้ว" }
		end
		ProfileAccess.Flush(player)
		return { true, ("ส่ง %s x%d ถึง %s แล้ว"):format(ItemCatalog.DisplayName(item.id, item.meta), count, friendName), item }
	end)
	Sending[player] = nil
	if not ok then
		warn("[Mail] send:", result)
		return false, "เกิดข้อผิดพลาด"
	end
	return result[1], result[2], result[3]
end

local function claim(player, letterId)
	local data = ProfileAccess.Get(player)
	if not data then return false, "ข้อมูลยังไม่พร้อม" end
	local _, letter = Mail.Find(data, letterId)
	if not letter then return false, "ไม่พบจดหมาย" end
	if letter.claimed then return false, "รับไปแล้ว" end

	-- ไอเทมที่ถูกถอดออกจากเกมหลังเมลถูกส่ง: ข้ามไป ไม่ให้ติดทั้งฉบับ
	local items, need = {}, {}
	for _, it in ipairs(letter.items) do
		if ItemCatalog.Exists(it.id) then
			items[#items + 1] = it
			need[it.id] = (need[it.id] or 0) + it.count
		end
	end
	-- รวมต่อชนิดก่อนถาม ไม่งั้นสองรายการชนิดเดียวกันผ่านด่านทีละอันแต่ใส่รวมกันไม่พอ
	-- ponytail: ของที่มี meta (สแตค 1) ถามรวมเป็นก้อนเดียวกับของชนิดเดียวกัน — เมลผู้เล่นมีของชิ้นเดียวจึงพอ
	for id, n in pairs(need) do
		if not Inventory.HasRoomFor(player, id, n) then return false, "กระเป๋าเต็ม เก็บของก่อนแล้วกดรับใหม่" end
	end

	letter.claimed = true
	letter.read = true
	for _, it in ipairs(items) do Inventory.AddItem(player, it.id, it.count, it.durability, it.meta) end
	if letter.gold > 0 then Inventory.AddGold(player, letter.gold) end
	if letter.exp > 0 then Progression.AwardExp(player, letter.exp) end
	ProfileAccess.Flush(player)
	return true, "รับของแล้ว"
end

local function deletable(letter)
	return letter.claimed or not Mail.HasAttachment(letter)
end

local function remove(player, letterId)
	local data = ProfileAccess.Get(player)
	if not data then return false, "ข้อมูลยังไม่พร้อม" end
	local index, letter = Mail.Find(data, letterId)
	if not letter then return false, "ไม่พบจดหมาย" end
	if not deletable(letter) then return false, "รับของก่อนแล้วค่อยลบ" end
	table.remove(Mail.Box(data), index)
	Mail.Drain(player, data)
	return true
end

local function clean(player)
	local data = ProfileAccess.Get(player)
	if not data then return false, "ข้อมูลยังไม่พร้อม" end
	local box = Mail.Box(data)
	for i = #box, 1, -1 do
		if deletable(box[i]) then table.remove(box, i) end
	end
	Mail.Drain(player, data)
	return true
end

local function markRead(player)
	local data = ProfileAccess.Get(player)
	if not data then return false end
	for _, letter in ipairs(Mail.Box(data)) do letter.read = true end
	return true
end

local OPS = {
	send = send, claim = claim, delete = remove, clean = clean, read = markRead,
	list = function() return true end,
}
